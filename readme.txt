- Edit the Backend/.env and Frontend/.env.local files according to your needs.
- To install the libraries, run the `npm install` commands in both the backend and frontend.
- For run project:
    backen -> "node index.js"
    frontend -> "npm run dev"
-Commands for adding tables to the database:

CREATE DATABASE INVOICE_APP;
USE INVOICE_APP;

CREATE TABLE USERS(
    USER_ID INT(10) UNSIGNED NOT NULL AUTO_INCREMENT,
    USER_NAME VARCHAR(256) NOT NULL UNIQUE,
    PASSWORD VARCHAR(1024) NOT NULL,
    RECORD_DATE DATETIME DEFAULT current_timestamp(),
    PRIMARY KEY (USER_ID)
) ENGINE=InnoDB;

CREATE TABLE REFRESH_TOKENS (
    USER_ID INT(10) UNSIGNED NOT NULL,
    TOKEN VARCHAR(512) NOT NULL,
    TOKEN_DATE DATETIME DEFAULT current_timestamp(),
    FOREIGN KEY (USER_ID)
        REFERENCES USERS(USER_ID)
        ON DELETE CASCADE
) ENGINE=InnoDB;

CREATE TABLE CUSTOMER (
    CUSTOMER_ID INT(10) UNSIGNED NOT NULL AUTO_INCREMENT,
    TAX_NUMBER VARCHAR(64) UNIQUE NOT NULL,
    TITLE VARCHAR(256),
    ADDRESS VARCHAR(512),
    EMAIL VARCHAR(256),
    USER_ID INT(10) UNSIGNED,
    RECORD_DATE DATETIME DEFAULT current_timestamp(),
    PRIMARY KEY (CUSTOMER_ID),
    FOREIGN KEY (USER_ID) REFERENCES USERS(USER_ID)
) ENGINE=InnoDB;

CREATE TABLE INVOICE (
    INVOICE_ID INT(10) UNSIGNED NOT NULL AUTO_INCREMENT,
    CUSTOMER_ID INT(10) UNSIGNED NOT NULL,
    INVOICE_NUMBER VARCHAR(128),
    INVOICE_DATE DATETIME,
    TOTAL_AMOUNT DECIMAL(10,2),
    USER_ID INT(10) UNSIGNED,
    RECORD_DATE DATETIME DEFAULT current_timestamp(),
    PRIMARY KEY (INVOICE_ID),
    FOREIGN KEY (CUSTOMER_ID) REFERENCES CUSTOMER(CUSTOMER_ID),
    FOREIGN KEY (USER_ID) REFERENCES USERS(USER_ID)
) ENGINE=InnoDB;

CREATE TABLE INVOICE_LINE (
    INVOICE_LINE_ID INT(10) UNSIGNED NOT NULL AUTO_INCREMENT,
    INVOICE_ID INT(10) UNSIGNED NOT NULL,
    ITEM_NAME VARCHAR(256),
    QUENTITY INT(10),
    PRICE DECIMAL(10,2),
    USER_ID INT(10) UNSIGNED,
    RECORD_DATE DATETIME DEFAULT current_timestamp(),
    PRIMARY KEY (INVOICE_LINE_ID),
    FOREIGN KEY (INVOICE_ID) REFERENCES INVOICE(INVOICE_ID),
    FOREIGN KEY (USER_ID) REFERENCES USERS(USER_ID)
) ENGINE=InnoDB;
